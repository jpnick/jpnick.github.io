การวิเคราะห์งบการเงินและมูลค่าของเงินตามเวลา

การเตรียมข้อมูลเบื้องต้น (Data Setup)

ตัวอย่างการคำนวณโดยใช้ Python สำหรับประมวลผลข้อมูลทางการเงินของบริษัท ABC

import pandas as pd
import numpy as np
from IPython.display import Markdown

# ข้อมูลจากงบการเงิน (หน่วย: ล้านบาท)
# Income Statement
sales = 120.0
cogs = 72.0
gross_profit = 48.0
ebit = 24.0
interest = 4.0
net_income = 16.0

# Balance Sheet
current_assets = 48.0
inventory = 24.0
total_assets = 120.0

current_liabilities = 20.0
total_liabilities = 50.0
total_equity = 70.0

เฉลยโจทย์ข้อที่ 1: การวิเคราะห์งบการเงินและการดำเนินงานธุรกิจ

คำถามข้อที่ 1.1: การคำนวณอัตราส่วนทางการเงิน (Financial Ratios)

1. อัตราส่วนสภาพคล่อง (Liquidity Ratios): - Current Ratio = \(\frac{\text{Current Assets}}{\text{Current Liabilities}}\) - Quick Ratio = \(\frac{\text{Current Assets} - \text{Inventory}}{\text{Current Liabilities}}\)

current_ratio = current_assets / current_liabilities
quick_ratio = (current_assets - inventory) / current_liabilities

print(f"Current Ratio = {current_ratio:.2f} เท่า")
print(f"Quick Ratio = {quick_ratio:.2f} เท่า")
Current Ratio = 2.40 เท่า
Quick Ratio = 1.20 เท่า

2. อัตราส่วนการบริหารสินทรัพย์ (Asset Management Ratios): - Inventory Turnover = \(\frac{\text{COGS}}{\text{Inventory}}\) - Days in Inventory = \(\frac{365}{\text{Inventory Turnover}}\)

inv_turnover = cogs / inventory
days_in_inventory = 365 / inv_turnover

print(f"Inventory Turnover = {inv_turnover:.2f} รอบ")
print(f"Days in Inventory = {days_in_inventory:.2f} วัน")
Inventory Turnover = 3.00 รอบ
Days in Inventory = 121.67 วัน

3. อัตราส่วนความสามารถในการชำระหนี้ (Financial Leverage Ratios): - Debt-to-Equity Ratio (D/E) = \(\frac{\text{Total Debt}}{\text{Total Equity}}\) - Total Debt Ratio = \(\frac{\text{Total Debt}}{\text{Total Assets}}\)

d_e_ratio = total_liabilities / total_equity
total_debt_ratio = total_liabilities / total_assets

print(f"D/E Ratio = {d_e_ratio:.3f} เท่า")
print(f"Total Debt Ratio = {total_debt_ratio:.3f} เท่า ({total_debt_ratio*100:.1f}%)")
D/E Ratio = 0.714 เท่า
Total Debt Ratio = 0.417 เท่า (41.7%)

4. อัตราส่วนความสามารถในการทำกำไร (Profitability Ratios): - Gross Profit Margin = \(\frac{\text{Gross Profit}}{\text{Sales}} \times 100\) - Operating Profit Margin = \(\frac{\text{EBIT}}{\text{Sales}} \times 100\) - Net Profit Margin = \(\frac{\text{Net Income}}{\text{Sales}} \times 100\) - Return on Assets (ROA) = \(\frac{\text{Net Income}}{\text{Total Assets}} \times 100\) - Return on Equity (ROE) = \(\frac{\text{Net Income}}{\text{Total Equity}} \times 100\)

gpm = (gross_profit / sales) * 100
opm = (ebit / sales) * 100
npm = (net_income / sales) * 100
roa = (net_income / total_assets) * 100
roe = (net_income / total_equity) * 100

print(f"Gross Profit Margin = {gpm:.2f}%")
print(f"Operating Profit Margin = {opm:.2f}%")
print(f"Net Profit Margin = {npm:.2f}%")
print(f"ROA = {roa:.2f}%")
print(f"ROE = {roe:.2f}%")
Gross Profit Margin = 40.00%
Operating Profit Margin = 20.00%
Net Profit Margin = 13.33%
ROA = 13.33%
ROE = 22.86%

คำถามข้อที่ 1.2: การวิเคราะห์ Dupont Analysis

สูตรของ DuPont Analysis 3 ปัจจัย คือ: \[\text{ROE} = \text{Net Profit Margin} \times \text{Asset Turnover} \times \text{Equity Multiplier}\]

asset_turnover = sales / total_assets
equity_multiplier = total_assets / total_equity
dupont_roe = (npm/100) * asset_turnover * equity_multiplier

print(f"1. Net Profit Margin = {npm:.2f}%")
print(f"2. Asset Turnover = {asset_turnover:.2f} เท่า")
print(f"3. Equity Multiplier = {equity_multiplier:.3f} เท่า")
print(f"ตรวจสอบ ROE จากสมการ = {dupont_roe * 100:.2f}%")
1. Net Profit Margin = 13.33%
2. Asset Turnover = 1.00 เท่า
3. Equity Multiplier = 1.714 เท่า
ตรวจสอบ ROE จากสมการ = 22.86%

การแปลความหมาย DUPONT Analysis:
- บริษัทมีความสามารถในการบริหารต้นทุนเพื่อทำกำไรได้ดี (Net Profit Margin อยู่ในระดับที่ดีที่ 13.33%) เมื่อเทียบกับ … ในธุรกิจเดียวกัน - การใช้สินทรัพย์ให้เกิดยอดขายทำได้อย่างมีประสิทธิภาพ (Asset Turnover = 1.0 เท่า) - ค่า Equity Multiplier ที่ 1.714 แสดงให้เห็นว่ามีหนี้สินอยู่ในระดับปานกลาง (Total Debt Ratio = 41.7%) ซึ่งถือเป็นระดับความเสี่ยงที่…

เฉลยโจทย์ข้อที่ 2: มูลค่าเงินตามเวลาและการประเมินโครงการ

คำถามข้อที่ 2.1: การวางแผนมูลค่าเงินอนาคตและมูลค่าปัจจุบัน

1) การลงทุนเพื่อขยายโรงงาน (Single Cash Flow): \[PV = \frac{FV}{(1+r)^n}\]

fv_target = 50000000
rate_inv = 0.08
years_inv = 5
pv_single = fv_target / ((1 + rate_inv)**years_inv)
print(f"PV: บริษัทต้องฝากเงินก้อนในวันนี้จำนวน = {pv_single:,.2f} บาท")
PV: บริษัทต้องฝากเงินก้อนในวันนี้จำนวน = 34,029,159.85 บาท

2) การจัดตั้งกองทุนเงินสะสม (Ordinary Annuity): \[PMT = \frac{FV}{\left[ \frac{(1+r)^n - 1}{r} \right]}\]

fv_factor = ((1 + rate_inv)**years_inv - 1) / rate_inv
pmt_annuity = fv_target / fv_factor
print(f"PMT: บริษัทต้องทยอยฝากเงินทุกสิ้นปี ปีละ = {pmt_annuity:,.2f} บาท")
PMT: บริษัทต้องทยอยฝากเงินทุกสิ้นปี ปีละ = 8,522,822.73 บาท

คำถามข้อที่ 2.2: การวิเคราะห์ตารางตัดจ่ายกู้ยืม (Loan Amortization)

1. คำนวณเงินงวดที่ต้องผ่อนชำระต่อปี (PMT): \[PMT = \frac{PV}{\left[ \frac{1 - (1+r)^{-n}}{r} \right]}\]

loan_pv = 10000000
loan_r = 0.06
loan_n = 4

pv_annuity_factor = (1 - (1 + loan_r)**-loan_n) / loan_r
pmt_loan = loan_pv / pv_annuity_factor
print(f"เงินผ่อนชำระเท่ากันทุกสิ้นปี (PMT) = {pmt_loan:,.2f} บาท")
เงินผ่อนชำระเท่ากันทุกสิ้นปี (PMT) = 2,885,914.92 บาท

คำถามข้อที่ 2.3: การตัดสินใจลงทุนในโครงการ (Capital Budgeting)

ข้อมูลกระแสเงินสด: \(CF_0 = -15m, CF_1 = 4m, CF_2 = 5m, CF_3 = 6m, CF_4 = 5m\) (WACC = 10%)

cf = np.array([-15000000, 4000000, 5000000, 6000000, 5000000])
discount_rate = 0.10

# 1. Payback Period (ระยะเวลาคืนทุนปกติ)
# สะสมกระแสเงินสด: ปี1(4m), ปี2(4+5=9m), ปี3(9+6=15m) -> คืนทุนปีที่ 3 พอดี
payback = 3.0

# 2. Discounted Payback Period (ระยะเวลาคืนทุนคิดลด)
discounted_cf = [c / (1 + discount_rate)**t for t, c in enumerate(cf)]
cumulative_dcf = np.cumsum(discounted_cf)

discounted_payback = 0.0
for i in range(1, len(cumulative_dcf)):
    if cumulative_dcf[i] >= 0:
        prev_cum = cumulative_dcf[i-1]
        curr_dcf = discounted_cf[i]
        discounted_payback = (i - 1) + abs(prev_cum) / curr_dcf
        break

# 3. Net Present Value (NPV)
npv = sum(discounted_cf)

# 4. Internal Rate of Return (IRR)
try:
    import numpy_financial as npf
    irr = npf.irr(cf)
except ImportError:
    # Manual estimation for fallback
    r_est = 0.0
    while sum(c / (1 + r_est)**t for t, c in enumerate(cf)) > 0:
        r_est += 0.0001
    irr = r_est

print(f"1. ระยะเวลาคืนทุน (Payback Period) = {payback:.2f} ปี")
print(f"2. ระยะเวลาคืนทุนคิดลด (Discounted Payback Period) = {discounted_payback:.2f} ปี")
print(f"3. มูลค่าปัจจุบันสุทธิ (NPV) = {npv:,.2f} บาท")
print(f"4. อัตราผลตอบแทนภายใน (IRR) = {irr*100:.2f}%")
1. ระยะเวลาคืนทุน (Payback Period) = 3.00 ปี
2. ระยะเวลาคืนทุนคิดลด (Discounted Payback Period) = 3.80 ปี
3. มูลค่าปัจจุบันสุทธิ (NPV) = 691,551.12 บาท
4. อัตราผลตอบแทนภายใน (IRR) = 12.02%

4. การตัดสินใจของบริษัท:
- บริษัท ควรอนุมัติ โครงการลงทุนนี้ เพราะ - พิจารณาจาก NPV: โครงการมีมูลค่าปัจจุบันสุทธิเป็นบวก (NPV = 691,556.76 บาท) ซึ่งหมายความว่าการลงทุนนี้สามารถสร้างมูลค่าเพิ่มให้กับบริษัทและผู้ถือหุ้นได้ - พิจารณาจาก IRR: อัตราผลตอบแทนภายในโครงการ (IRR = 11.87%) ซึ่งสูงกว่าต้นทุนทางการเงินหรือ WACC ของบริษัท (10.0%)
